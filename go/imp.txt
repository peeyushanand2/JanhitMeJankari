func addTable(wg *sync.WaitGroup) {
    // Schedule the call to WaitGroup's Done to tell goroutine is completed.
    defer wg.Done()
    var wgaddTable sync.WaitGroup
    for i := 1; i <= 10; i++ {
       wgaddTable.Add(1)
       go func() {
          defer wgaddTable.Done()
          // do the code
       }()
       sleep := rand.Int63n(1000)
       time.Sleep(time.Duration(sleep) * time.Millisecond)
       fmt.Println("Addition Table for:", i)
       for j := 1; j <= 10; j++ {
          //res = i + j
          fmt.Printf("%d+%d=%d\t", i, j, i+j)
       }
       fmt.Println("\n")
    }
    wgaddTable.Wait()
}

"Pipeline" Concurrency Pattern:
 
Problem: Print out fibonacci values of 
10 randomly generated numbers
 
Pipeline with three Goroutines:
 
Goroutine 1: Find out 10 random numbers
Goroutine 2: Find out fibonacci values for the value 
             sent by Goroutine 1
Goroutine 3: Print out fibonacci values for the value 
             sent by Goroutine 2
